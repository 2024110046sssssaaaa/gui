from flask import Flask, request, jsonify
import os, sqlite3

app = Flask(__name__)
DB_PATH = os.environ.get('DB_PATH', '/data/app.db')

@app.route('/api/invoices', methods=['GET'])
def get_invoices():
    conn = sqlite3.connect(DB_PATH)
    cur = conn.cursor()
    rows = cur.execute('SELECT * FROM invoices').fetchall()
    return jsonify(rows)
